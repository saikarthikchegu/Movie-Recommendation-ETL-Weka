import pandas as pd

# 1. Load the original CSV files
ratings_path = r'C:\PentahoProjects\ml-latest-small\ratings.csv'
movies_path = r'C:\PentahoProjects\ml-latest-small\movies.csv'

ratings = pd.read_csv(ratings_path)
movies = pd.read_csv(movies_path)

# 2. Perform Grouping and Aggregation on ratings
agg_ratings = ratings.groupby('movieId').agg(
    avg_rating=('rating', 'mean'),
    total_ratings=('rating', 'count')
).reset_index()

# 3. Join aggregated ratings with movies data
merged_df = pd.merge(movies, agg_ratings, on='movieId', how='inner')

# 4. Calculate custom recommendation score
merged_df['recommendation_score'] = merged_df['avg_rating'] * merged_df['total_ratings']

# 5. Save output file ready for WEKA
output_path = r'C:\PentahoProjects\ml-latest-small\weka_ready_movies.csv'
merged_df.to_csv(output_path, index=False)

print("Data processing complete! Output saved to:", output_path)
